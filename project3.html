sales_tidy <- sales |>
    ## Removes "extra stuff" from datesm then convert use stringr to get date (includes regular expression)
    mutate(
        released = str_remove(released, " \\([A-Z]+\\)\\[[0-9]+\\]"),
        released = mdy(released)
    ) |>
    ## Use forcats to create a factor called country 
    mutate(country = fct_collapse(country,
        "World" = c("World", "WW"),
        "FRA"   = c("FRA", "FR")
    )) |>
    ## Convert sales to album sales in millions of dollars.
    mutate(sales = sales / 1000000) |>
    ## Keep only album sales from the UK, the US or the World
    filter(country %in% c("UK", "US", "World"))

## Completing Parts 1-6 
ts_album_sentiment <- ts_lyrics %>%
    ## Clean album names (remove white space at end) so they match the sales data (source below)
    mutate(Album = str_to_title(str_trim(Album))) %>%
    ## 1. One word per row
    unnest_tokens(output = word, input = Lyrics, token = "words") %>%
    ## 2. Removing stop words
    anti_join(stop_words, by = "word") %>%
    ## 3. Calculate the total number for each word in the lyrics for each Album
    count(Album, word) %>%
    ## 4. Add column with afinn sentiment score 
    inner_join(get_sentiments("afinn"), by = "word") %>%
    ## 5. Calculate the average sentiment score for each album, weighted by how often each word appears
    group_by(Album) %>%
    summarize(avg_sentiment = sum(value * n) / sum(n))
